if(MOZART_WITH_ORT)
    find_library(ENHANCE_ORT_LIBRARY onnxruntime HINTS "${ONNX_RUNTIME_PATH}" /usr/local/lib /usr/lib/aarch64-linux-gnu)
    find_path(ENHANCE_ORT_INCLUDE onnxruntime_cxx_api.h HINTS "${ONNX_RUNTIME_INCLUDE_DIR}" /usr/local/include /usr/include/onnxruntime)
    if(NOT ENHANCE_ORT_LIBRARY OR NOT ENHANCE_ORT_INCLUDE)
        message(FATAL_ERROR "enhance requires ONNX Runtime headers and library")
    endif()
    target_compile_definitions(mozart_enhance PRIVATE MOZART_ENHANCE_ORT=1)
    target_include_directories(mozart_enhance SYSTEM PRIVATE "${ENHANCE_ORT_INCLUDE}")
    target_link_libraries(mozart_enhance PRIVATE "${ENHANCE_ORT_LIBRARY}")
endif()
